cmake_minimum_required(VERSION 3.16)
project(lane_map)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(vision_lane_interfaces REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
add_executable(map_fusion_node src/map_fusion_node.cpp)
target_include_directories(map_fusion_node PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(map_fusion_node rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs visualization_msgs vision_lane_interfaces)
target_link_libraries(map_fusion_node ${OpenCV_LIBS})
target_compile_options(map_fusion_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS map_fusion_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_line_bank test/test_line_bank.cpp)
  target_include_directories(test_line_bank PRIVATE include)
  target_compile_options(test_line_bank PRIVATE -UNDEBUG)
  add_test(NAME test_line_bank COMMAND test_line_bank)
  add_executable(test_recovery_identity test/test_recovery_identity.cpp)
  target_include_directories(test_recovery_identity PRIVATE include)
  target_compile_options(test_recovery_identity PRIVATE -UNDEBUG)
  add_test(NAME test_recovery_identity COMMAND test_recovery_identity)
  add_executable(test_registration test/test_registration.cpp)
  target_include_directories(test_registration PRIVATE include)
  target_compile_options(test_registration PRIVATE -UNDEBUG)
  add_test(NAME test_registration COMMAND test_registration)
endif()
ament_package()
