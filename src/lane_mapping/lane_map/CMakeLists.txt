cmake_minimum_required(VERSION 3.16)
project(lane_map)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(vision_lane_interfaces REQUIRED)
find_package(monoscale_pose REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
add_executable(map_fusion_node src/map_fusion_node.cpp)
target_include_directories(map_fusion_node PRIVATE include ${OpenCV_INCLUDE_DIRS} ${monoscale_pose_INCLUDE_DIRS})
ament_target_dependencies(map_fusion_node rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs visualization_msgs vision_lane_interfaces)
target_link_libraries(map_fusion_node ${OpenCV_LIBS})
target_compile_options(map_fusion_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS map_fusion_node DESTINATION lib/${PROJECT_NAME})
# E-8: golden output of the observation path, built against the node source (a tool, not
# a test: its inputs come from a recorded run).
add_executable(lane_map_apply_golden test/lane_map_apply_golden.cpp)
target_include_directories(lane_map_apply_golden PRIVATE include ${OpenCV_INCLUDE_DIRS} ${monoscale_pose_INCLUDE_DIRS})
ament_target_dependencies(lane_map_apply_golden rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs visualization_msgs vision_lane_interfaces)
target_link_libraries(lane_map_apply_golden ${OpenCV_LIBS})
install(TARGETS lane_map_apply_golden DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
# Header-only lane tracking/mapping shared with lane_detection.
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(monoscale_pose)
if(BUILD_TESTING)
