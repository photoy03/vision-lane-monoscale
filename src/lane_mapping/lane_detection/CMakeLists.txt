cmake_minimum_required(VERSION 3.16)
project(lane_detection)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)

find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(vision_lane_interfaces REQUIRED)
find_package(monoscale_pose REQUIRED)
find_package(lane_map REQUIRED)
find_package(OpenCV REQUIRED
  COMPONENTS core imgproc imgcodecs calib3d videoio
  OPTIONAL_COMPONENTS cudaarithm cudaimgproc cudawarping cudafilters)

add_executable(global_mapping_node src/global_mapping_node.cpp)
ament_target_dependencies(global_mapping_node
  rclcpp sensor_msgs nav_msgs std_msgs geometry_msgs cv_bridge ament_index_cpp
  vision_lane_interfaces)
target_link_libraries(global_mapping_node ${OpenCV_LIBS})

add_executable(compressed_image_replay_node src/compressed_image_replay_node.cpp)
ament_target_dependencies(compressed_image_replay_node rclcpp sensor_msgs)
target_link_libraries(compressed_image_replay_node ${OpenCV_LIBS})
target_include_directories(compressed_image_replay_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_options(compressed_image_replay_node PRIVATE -O3 -DNDEBUG)
target_include_directories(global_mapping_node PRIVATE include ${OpenCV_INCLUDE_DIRS} ${monoscale_pose_INCLUDE_DIRS} ${lane_map_INCLUDE_DIRS})
target_compile_options(global_mapping_node PRIVATE -O3 -DNDEBUG)

set(VISION_LANE_OPENCV_CUDA_FOUND TRUE)
foreach(cuda_component cudaarithm cudaimgproc cudawarping cudafilters)
  if(NOT OpenCV_${cuda_component}_FOUND AND NOT TARGET opencv_${cuda_component})
    set(VISION_LANE_OPENCV_CUDA_FOUND FALSE)
  endif()
endforeach()

if(VISION_LANE_OPENCV_CUDA_FOUND)
  target_compile_definitions(global_mapping_node PRIVATE VISION_LANE_HAS_OPENCV_CUDA=1)
  foreach(cuda_component cudaarithm cudaimgproc cudawarping cudafilters)
    if(TARGET opencv_${cuda_component})
      target_link_libraries(global_mapping_node opencv_${cuda_component})
    endif()
  endforeach()
  message(STATUS "vision_lane: OpenCV CUDA preprocessing enabled")
else()
  target_compile_definitions(global_mapping_node PRIVATE VISION_LANE_HAS_OPENCV_CUDA=0)
  message(WARNING "vision_lane: OpenCV CUDA modules not found; CPU backend only")
endif()

install(TARGETS global_mapping_node compressed_image_replay_node DESTINATION lib/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
  add_executable(test_pending_lane_observations test/test_pending_lane_observations.cpp)
  target_include_directories(test_pending_lane_observations PRIVATE src ${monoscale_pose_INCLUDE_DIRS} ${lane_map_INCLUDE_DIRS})
  target_compile_options(test_pending_lane_observations PRIVATE -UNDEBUG)
  add_test(NAME test_pending_lane_observations COMMAND test_pending_lane_observations)
  add_executable(test_pending_readiness test/test_pending_readiness.cpp)
  target_include_directories(test_pending_readiness PRIVATE src ${monoscale_pose_INCLUDE_DIRS} ${lane_map_INCLUDE_DIRS})
  target_compile_options(test_pending_readiness PRIVATE -UNDEBUG)
  add_test(NAME test_pending_readiness COMMAND test_pending_readiness)
  add_executable(test_startup_warmup test/test_startup_warmup.cpp)
  target_include_directories(test_startup_warmup PRIVATE src)
  target_compile_options(test_startup_warmup PRIVATE -UNDEBUG)
  add_test(NAME test_startup_warmup COMMAND test_startup_warmup)
endif()

install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/convert_fisheye_npz.py scripts/compare_profiles.py
  scripts/compare_preprocess_backends.py
  scripts/plot_profiles.py
  DESTINATION lib/${PROJECT_NAME})
# E-8: the detection code as a header, and its golden-output check (tools, not a test:
# it needs recorded frames that do not ship with the package).
install(DIRECTORY include/ DESTINATION include)
add_executable(lane_pipeline_golden test/lane_pipeline_golden.cpp)
target_include_directories(lane_pipeline_golden PRIVATE include ${OpenCV_INCLUDE_DIRS} ${lane_map_INCLUDE_DIRS})
target_link_libraries(lane_pipeline_golden ${OpenCV_LIBS})
target_compile_options(lane_pipeline_golden PRIVATE -O3 -DNDEBUG)
install(TARGETS lane_pipeline_golden RUNTIME DESTINATION lib/${PROJECT_NAME})
add_executable(lane_offline src/lane_offline.cpp)
target_include_directories(lane_offline PRIVATE include src ${OpenCV_INCLUDE_DIRS} ${lane_map_INCLUDE_DIRS} ${monoscale_pose_INCLUDE_DIRS})
target_link_libraries(lane_offline ${OpenCV_LIBS})
target_compile_options(lane_offline PRIVATE -O3 -DNDEBUG)
install(TARGETS lane_offline RUNTIME DESTINATION lib/${PROJECT_NAME})

ament_package()
